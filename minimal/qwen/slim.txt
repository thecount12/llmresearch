// Function to generate a random 32-bit unsigned integer
unsigned int random_u32(uint64_t* state) {
    *state ^= *state >> 12;
    *state ^= *state << 25;
    *state ^= *state >> 27;
    return (*state * 0x2545F4914F6CDD1Dull) >> 32;
}

// Function to generate a random 32-bit float in the range [0, 1)
float random_f32(uint64_t* state) {
    return (random_u32(state) >> 8) / 16777216.0f;
}

// Function to sample an index from a probability distribution
int sample_mult(float* probabilities, int n, float coin) {
    float cdf = 0.0f;
    for (int i = 0; i < n; i++) {
        cdf += probabilities[i];
        if (coin < cdf) {
            return i;
        }
    }
    return n - 1; // in case of rounding errors
}

// Function to safely print a string to standard output
void safe_printf(const char* str) {
    if (str == NULL) return;
    fputs(str, stdout);
}

// Function to get the current time in seconds
void clock_gettime(struct timespec *ts) {

}

// Function to print a formatted string (similar to printf)
void printf(const char* format, ...) {
    va_list ap;
    va_start(ap, format);
    vprintf(format, ap);
    va_end(ap);
}
